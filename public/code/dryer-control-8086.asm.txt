;====================================================================
; TITLE: FINAL CONTROLLER PROJECT  FINAL VERSION (NO PORT CONTENTION)
; DESCRIPTION:
; 8086-based dryer control system with temperature and pressure control
;====================================================================

.MODEL SMALL
.STACK 100H

.DATA
; ---------------- I/O PORT ADDRESSES ----------------
PORTA_ADDR     EQU 00H      ; 7-segment display output
PORTB_ADDR     EQU 02H      ; ADC data input
PORTC_ADDR     EQU 04H      ; Buttons + control outputs
CTRL_ADDR      EQU 06H      ; 8255 control register

; ---------------- SYSTEM VARIABLES ----------------
TEMP_SETPOINT  DB 0         ; Temperature setpoint
PRESS_SETPOINT DB 0         ; Pressure setpoint
TEMP_ACTUAL   DB 0         ; Actual temperature value
PRESS_ACTUAL  DB 0         ; Actual pressure value

MENU_MODE      DB 0         ; 0 = Temperature mode, 1 = Pressure mode
RUN_STATE      DB 0         ; 0 = STOP, 1 = RUN
OUT_IMAGE      DB 03H       ; Output image for Port C

.CODE
MAIN PROC
; Initialize data segment
MOV AX, @DATA
MOV DS, AX

; Initialize 8255 PPI in Mode 0
MOV AL, 8AH
OUT CTRL_ADDR, AL

; Reset outputs
MOV AL, 03H
MOV OUT_IMAGE, AL
OUT PORTC_ADDR, AL

; Display initial setpoint
CALL DISPLAY_SETPOINT

; ================= MAIN LOOP =================
MAIN_LOOP:
IN AL, PORTC_ADDR
MOV BL, AL

TEST BL, 10000000B
JZ HANDLE_START

TEST BL, 00010000B
JZ HANDLE_MODE

TEST BL, 00100000B
JZ HANDLE_UP

TEST BL, 01000000B
JZ HANDLE_DOWN

CMP RUN_STATE, 1
JE RUN_MODE

; ---------------- STOP MODE ----------------
MOV AL, OUT_IMAGE
AND AL, 11110011B
MOV OUT_IMAGE, AL
CALL SEND_OUTPUT
CALL DISPLAY_SETPOINT
JMP MAIN_LOOP

; ================= RUN MODE =================
RUN_MODE:
IN AL, PORTC_ADDR
TEST AL, 00010000B
JZ HANDLE_MODE

CALL READ_TEMP_SENSOR
CALL READ_PRESS_SENSOR

; Heater control
MOV AL, TEMP_ACTUAL
CMP AL, TEMP_SETPOINT
JB TURN_ON_HEATER
AND OUT_IMAGE, 11111011B
JMP CHECK_ALARM

TURN_ON_HEATER:
OR OUT_IMAGE, 00000100B

CHECK_ALARM:
MOV AL, PRESS_ACTUAL
CMP AL, PRESS_SETPOINT
JA TURN_ON_BUZZER
AND OUT_IMAGE, 11110111B
JMP UPDATE_OUTPUT

TURN_ON_BUZZER:
OR OUT_IMAGE, 00001000B

UPDATE_OUTPUT:
CALL SEND_OUTPUT
CALL DISPLAY_ACTUAL
JMP MAIN_LOOP

; ================= BUTTON HANDLERS =================
HANDLE_START:
CALL DELAY_DEBOUNCE
XOR RUN_STATE, 1
MOV AL, 00H
OUT PORTA_ADDR, AL
CALL DELAY_BLINK
JMP WAIT_RELEASE

HANDLE_MODE:
CALL DELAY_DEBOUNCE
XOR MENU_MODE, 1
JMP WAIT_RELEASE

HANDLE_UP:
CALL DELAY_DEBOUNCE
CMP RUN_STATE, 1
JE MAIN_LOOP
CMP MENU_MODE, 0
JE INC_TEMP
INC PRESS_SETPOINT
JMP WAIT_RELEASE

INC_TEMP:
INC TEMP_SETPOINT
JMP WAIT_RELEASE

HANDLE_DOWN:
CALL DELAY_DEBOUNCE
CMP RUN_STATE, 1
JE MAIN_LOOP
CMP MENU_MODE, 0
JE DEC_TEMP
DEC PRESS_SETPOINT
JMP WAIT_RELEASE

DEC_TEMP:
DEC TEMP_SETPOINT
JMP WAIT_RELEASE

WAIT_RELEASE:
IN AL, PORTC_ADDR
AND AL, 11110000B
CMP AL, 11110000B
JNE WAIT_RELEASE
JMP MAIN_LOOP

; ================= OUTPUT =================
SEND_OUTPUT PROC
MOV AL, OUT_IMAGE
AND AL, 00001111B
OUT PORTC_ADDR, AL
RET
SEND_OUTPUT ENDP

; ================= ADC =================
READ_TEMP_SENSOR PROC
AND OUT_IMAGE, 11111110B
CALL SEND_OUTPUT
MOV CX, 100
D1: LOOP D1
IN AL, PORTB_ADDR
MOV TEMP_ACTUAL, AL
OR OUT_IMAGE, 00000001B
CALL SEND_OUTPUT
RET
READ_TEMP_SENSOR ENDP

READ_PRESS_SENSOR PROC
AND OUT_IMAGE, 11111101B
CALL SEND_OUTPUT
MOV CX, 100
D2: LOOP D2
IN AL, PORTB_ADDR
MOV PRESS_ACTUAL, AL
OR OUT_IMAGE, 00000010B
CALL SEND_OUTPUT
RET
READ_PRESS_SENSOR ENDP

; ================= DISPLAY =================
DISPLAY_SETPOINT PROC
CMP MENU_MODE, 0
JE DSP_TEMP
MOV AL, PRESS_SETPOINT
JMP CONV1
DSP_TEMP:
MOV AL, TEMP_SETPOINT
CONV1:
CALL CONVERT_BCD
RET
DISPLAY_SETPOINT ENDP

DISPLAY_ACTUAL PROC
CMP MENU_MODE, 0
JE DSPA_TEMP
MOV AL, PRESS_ACTUAL
JMP CONV2
DSPA_TEMP:
MOV AL, TEMP_ACTUAL
CONV2:
CALL CONVERT_BCD
RET
DISPLAY_ACTUAL ENDP

; ================= BCD =================
CONVERT_BCD PROC
PUSH CX
PUSH BX
PUSH DX
MOV AH, 0
MOV BL, 10
DIV BL
MOV CL, 4
SHL AL, CL
OR AL, AH
OUT PORTA_ADDR, AL
POP DX
POP BX
POP CX
RET
CONVERT_BCD ENDP

; ================= DELAYS =================
DELAY_DEBOUNCE PROC
MOV CX, 2000H
L1: LOOP L1
RET
DELAY_DEBOUNCE ENDP

DELAY_BLINK PROC
MOV CX, 0FFFFH
L2: LOOP L2
RET
DELAY_BLINK ENDP

END MAIN
